import { env } from 'cloudflare:workers';
const kinds=['transaction','account','budget','task'];
function db(){if(!env.DB)throw new Error('Storage unavailable');return env.DB;}
function valid(x:any){if(!x||!kinds.includes(x.kind)||!x.data||typeof x.data.title!=='string'||!x.data.title.trim()||x.data.title.length>200)throw new Error('Please enter a valid name.');const d=x.data;if(!['Personal','Business'].includes(d.mode))throw new Error('Choose a mode.');if(x.kind!=='task'&&(!Number.isSafeInteger(d.amount)||d.amount<0||d.amount>1e12))throw new Error('Enter a valid amount.');if(['transaction','task'].includes(x.kind)&&!/^\d{4}-\d{2}-\d{2}$/.test(d.date))throw new Error('Choose a date.');if(x.kind==='transaction'&&(!['income','expense'].includes(d.type)||!['Food','Transport','Bills','Shopping','Health','Work','Other'].includes(d.category)))throw new Error('Choose a category and type.');if(x.kind==='budget'&&!/^\d{4}-\d{2}$/.test(d.month))throw new Error('Choose a month.');return x;}
function fail(e:unknown,status=400){return Response.json({error:e instanceof Error?e.message:'Unable to save'}, {status});}
export async function GET(){try{const r=await db().prepare('SELECT * FROM records ORDER BY created DESC').all();return Response.json({records:r.results.map((r:any)=>({...r,data:JSON.parse(r.data)}))});}catch{return fail(new Error('Your data could not load. Please retry.'),503);}}
export async function POST(req:Request){try{const x=valid(await req.json());const id=crypto.randomUUID();await db().prepare('INSERT INTO records (id,kind,data,created) VALUES (?,?,?,?)').bind(id,x.kind,JSON.stringify(x.data),Date.now()).run();return Response.json({id});}catch(e){return fail(e);}}
export async function PUT(req:Request){try{const x=valid(await req.json());if(typeof x.id!=='string')throw new Error('Invalid record');await db().prepare('UPDATE records SET data=? WHERE id=? AND kind=?').bind(JSON.stringify(x.data),x.id,x.kind).run();return Response.json({ok:true});}catch(e){return fail(e);}}
export async function DELETE(req:Request){try{const {id}=await req.json() as {id:string};if(typeof id!=='string')throw new Error('Invalid record');await db().prepare('DELETE FROM records WHERE id=?').bind(id).run();return Response.json({ok:true});}catch(e){return fail(e);}}
